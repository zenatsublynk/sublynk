// ─── Sublynk GC Onboarding Intake - Apps Script Backend ───
// Deploy as: Web App > Execute as Me > Anyone can access

var PARENT_FOLDER_ID = '1KQxjb19o84Az5TTaRoOahcPfS30JfLKK';
// Set these in Apps Script: Project Settings > Script Properties
// NOTION_TOKEN = your Notion integration token
// NOTION_DB_ID = your onboarding database ID
var NOTION_TOKEN = PropertiesService.getScriptProperties().getProperty('NOTION_TOKEN');
var NOTION_DB_ID = PropertiesService.getScriptProperties().getProperty('NOTION_DB_ID');

// ─── API endpoint for external calls (GitHub Pages, etc.) ───
function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents);
    var action = body.action;
    var result;

    switch (action) {
      case 'processSubmission':
        result = processSubmission(body.payload);
        break;
      case 'createOnboardingFolder':
        result = createOnboardingFolder(body.companyName, body.dateStr, body.xlsxBase64, body.fileName);
        break;
      case 'uploadContractFile':
        result = { fileId: uploadContractFile(body.base64Data, body.fileName, body.companyName) };
        break;
      case 'uploadCoiSample':
        result = { url: uploadCoiSample(body.base64Data, body.fileName, body.companyName, body.contactName) };
        break;
      default:
        result = { error: 'Unknown action: ' + action };
    }

    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ─── Serve the form ───
function doGet() {
  return HtmlService.createHtmlOutputFromFile('intake')
    .setTitle('Sublynk - GC Onboarding Intake')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

// ─── Create GC folder structure + upload XLSX ───
// Called on form submit. Creates:
//   {Parent}/
//     {Company Name}/
//       Network Setup Requirements {Company} {Date}.xlsx
//       COI Sample/
//       Contract Sample/
function createOnboardingFolder(companyName, dateStr, xlsxBase64, fileName) {
  var parentFolder = DriveApp.getFolderById(PARENT_FOLDER_ID);

  // Create or find company folder
  var companyFolders = parentFolder.getFoldersByName(companyName);
  var companyFolder;
  if (companyFolders.hasNext()) {
    companyFolder = companyFolders.next();
  } else {
    companyFolder = parentFolder.createFolder(companyName);
  }

  // Create subfolders if they don't exist
  ['COI Sample', 'Contract Sample'].forEach(function(name) {
    var existing = companyFolder.getFoldersByName(name);
    if (!existing.hasNext()) {
      companyFolder.createFolder(name);
    }
  });

  // Upload XLSX (replace if exists)
  var xlsxBlob = Utilities.newBlob(
    Utilities.base64Decode(xlsxBase64),
    'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
    fileName
  );

  var existingFiles = companyFolder.getFilesByName(fileName);
  if (existingFiles.hasNext()) {
    existingFiles.next().setTrashed(true);
  }

  var file = companyFolder.createFile(xlsxBlob);

  return {
    folderId: companyFolder.getId(),
    folderUrl: companyFolder.getUrl(),
    fileId: file.getId(),
    fileUrl: file.getUrl()
  };
}

// ─── Upload contract file to Drive ───
function uploadContractFile(base64Data, fileName, companyName) {
  var parentFolder = DriveApp.getFolderById(PARENT_FOLDER_ID);

  // Find or create company folder
  var companyFolders = parentFolder.getFoldersByName(companyName);
  var companyFolder;
  if (companyFolders.hasNext()) {
    companyFolder = companyFolders.next();
  } else {
    companyFolder = parentFolder.createFolder(companyName);
  }

  // Find or create Contract Sample subfolder
  var contractFolders = companyFolder.getFoldersByName('Contract Sample');
  var contractFolder;
  if (contractFolders.hasNext()) {
    contractFolder = contractFolders.next();
  } else {
    contractFolder = companyFolder.createFolder('Contract Sample');
  }

  var blob = Utilities.newBlob(
    Utilities.base64Decode(base64Data),
    'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
    fileName
  );
  var file = contractFolder.createFile(blob);
  return file.getId();
}

// ─── Upload COI sample to Drive ───
function uploadCoiSample(base64Data, fileName, companyName, contactName) {
  var parentFolder = DriveApp.getFolderById(PARENT_FOLDER_ID);

  // Find or create company folder
  var companyFolders = parentFolder.getFoldersByName(companyName);
  var companyFolder;
  if (companyFolders.hasNext()) {
    companyFolder = companyFolders.next();
  } else {
    companyFolder = parentFolder.createFolder(companyName);
  }

  // Find or create COI Sample subfolder
  var coiFolders = companyFolder.getFoldersByName('COI Sample');
  var coiFolder;
  if (coiFolders.hasNext()) {
    coiFolder = coiFolders.next();
  } else {
    coiFolder = companyFolder.createFolder('COI Sample');
  }

  var mimeType = 'application/pdf';
  var lower = fileName.toLowerCase();
  if (lower.match(/\.png$/)) mimeType = 'image/png';
  else if (lower.match(/\.jpe?g$/)) mimeType = 'image/jpeg';
  else if (lower.match(/\.docx?$/)) mimeType = 'application/vnd.openxmlformats-officedocument.wordprocessingml.document';

  var blob = Utilities.newBlob(
    Utilities.base64Decode(base64Data),
    mimeType,
    fileName
  );
  var file = coiFolder.createFile(blob);
  return file.getUrl();
}

// ─── Process full submission ───
function processSubmission(payload) {
  // Log to spreadsheet
  logToSheet(payload);

  // Create folder structure if not already done
  var folderUrl = '';
  try {
    var parentFolder = DriveApp.getFolderById(PARENT_FOLDER_ID);
    var companyName = payload.legalName || 'Unknown';
    var companyFolders = parentFolder.getFoldersByName(companyName);
    var companyFolder;
    if (companyFolders.hasNext()) {
      companyFolder = companyFolders.next();
    } else {
      companyFolder = parentFolder.createFolder(companyName);
      companyFolder.createFolder('COI Sample');
      companyFolder.createFolder('Contract Sample');
    }
    folderUrl = companyFolder.getUrl();
  } catch(e) {
    Logger.log('Folder creation in processSubmission failed: ' + e);
  }

  // Post to Notion
  try {
    postToNotion(payload, folderUrl);
  } catch(e) {
    Logger.log('Notion post failed: ' + e);
  }

  return { success: true };
}

// ─── NOTION: Create onboarding entry ───
function postToNotion(data, folderUrl) {
  var insurance = [];
  if (data.cglEnabled) insurance.push({name: 'CGL'});
  if (data.pollutionEnabled) insurance.push({name: 'Pollution'});
  if (data.autoEnabled) insurance.push({name: 'Auto'});
  if (data.wcEnabled) insurance.push({name: 'Workers Comp'});
  if (data.umbrellaEnabled) insurance.push({name: 'Umbrella'});
  if (data.plEnabled) insurance.push({name: 'Professional'});
  if (data.baileesEnabled) insurance.push({name: 'Bailees'});

  var programs = (data.selectedPrograms || data.programs || []).map(function(p) { return {name: p}; });

  var payload = {
    parent: { database_id: NOTION_DB_ID },
    properties: {
      'Company Name': { title: [{text: {content: data.legalName || 'Unknown'}}] },
      'Status': { select: {name: 'New'} },
      'Submitted': { date: {start: Utilities.formatDate(new Date(), 'America/Los_Angeles', 'yyyy-MM-dd')} },
      'Primary Email': { email: data.email || null },
      'City': { rich_text: [{text: {content: data.city || ''}}] },
      'State': { rich_text: [{text: {content: data.state || ''}}] },
      'Programs': { multi_select: programs },
      'Insurance Types': { multi_select: insurance },
      'BG Checks': { checkbox: data.bgChecksRequired ? true : false }
    }
  };

  if (folderUrl) {
    payload.properties['Contract Folder'] = { url: folderUrl };
  }

  var resp = UrlFetchApp.fetch('https://api.notion.com/v1/pages', {
    method: 'post',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + NOTION_TOKEN, 'Notion-Version': '2022-06-28' },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });

  var result = JSON.parse(resp.getContentText());
  if (result.id) {
    Logger.log('Notion page created: ' + result.id);
  } else {
    Logger.log('Notion error: ' + resp.getContentText());
  }
  return result;
}

// ─── Log submission to Google Sheet ───
function logToSheet(payload) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) {
    // If no bound spreadsheet, create one in the parent folder
    var parentFolder = DriveApp.getFolderById(PARENT_FOLDER_ID);
    ss = SpreadsheetApp.create('Sublynk Onboarding Submissions');
    var file = DriveApp.getFileById(ss.getId());
    parentFolder.addFile(file);
    DriveApp.getRootFolder().removeFile(file);
  }

  var sheet = ss.getSheetByName('Submissions');
  if (!sheet) {
    sheet = ss.insertSheet('Submissions');
    // Add headers on first run
    var headers = [
      'Timestamp', 'Legal Business Name', 'DBA', 'Tax ID', 'Tax Classification',
      'Website', 'Address', 'City', 'State', 'Zip', 'Primary Email',
      'Certificate Holder', 'Description of Operations',
      'Signatory Name', 'Signatory Title', 'Signatory Email',
      'Counter Signatures', 'Additional Contracts', 'Drive File IDs'
    ];
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold');
  }

  var row = [
    new Date(),
    payload.legalName || '',
    payload.dba || '',
    payload.taxId || '',
    payload.taxClass || '',
    payload.website || '',
    payload.address1 || '',
    payload.city || '',
    payload.state || '',
    payload.zip || '',
    payload.primaryEmail || '',
    payload.certHolder || '',
    payload.descOps || '',
    payload.signatoryName || '',
    payload.signatoryTitle || '',
    payload.signatoryEmail || '',
    payload.counterSigs || '',
    payload.additionalContracts || '',
    (payload.driveFileIds || []).join(', ')
  ];

  sheet.appendRow(row);
}
